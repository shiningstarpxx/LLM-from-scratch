## 2.1 训练内存公式

```
训练内存 = 模型参数 + 梯度 + 优化器状态 + 激活值

详细分解 (混合精度 + Adam):
- FP32 主权重: 4P bytes
- FP16 工作权重: 2P bytes
- FP16 梯度: 2P bytes
- Adam 动量 (m): 4P bytes
- Adam 方差 (v): 4P bytes
─────────────────────────
固定开销: 16P bytes

+ 激活值: 变化大，依赖batch/seq

其中 P = 参数数量
```

---

## 2.2 为什么需要FP32主权重？

```python
# 混合精度训练的核心流程

# 1. 维护FP32主权重 (高精度)
master_weights = model.parameters()  # FP32, 28GB for 7B

# 2. 复制到FP16进行计算
working_weights = master_weights.half()  # FP16, 14GB

# 3. FP16前向传播
with autocast():  # 自动混合精度
    output = model(input)  # 使用working_weights
    loss = criterion(output, target)

# 4. FP16反向传播
scaler.scale(loss).backward()  # 梯度也是FP16

# 5. 更新FP32主权重
scaler.step(optimizer)  # optimizer更新FP32权重

# 为什么FP32必须？
# FP16范围: 6×10⁻⁵ ~ 65504
# 学习率: 1e-4, 梯度: 1e-5
# 更新量: 1e-9 → 超出FP16精度!
```

## 2.4 Adam优化器状态详解

```python
# Adam算法
def adam_step(param, grad, m, v, lr, beta1=0.9, beta2=0.999, eps=1e-8, t=1):
    """
    m: 一阶矩 (动量), 跟踪梯度的指数移动平均
    v: 二阶矩 (方差), 跟踪梯度平方的指数移动平均
    """
    # 更新动量
    m = beta1 * m + (1 - beta1) * grad

    # 更新方差
    v = beta2 * v + (1 - beta2) * grad ** 2

    # 偏差校正
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)

    # 更新参数
    param = param - lr * m_hat / (torch.sqrt(v_hat) + eps)

    return param, m, v

# 内存需求:
# - m: 与参数同大小, FP32, 4 bytes/param
# - v: 与参数同大小, FP32, 4 bytes/param
# 7B模型: (4 + 4) × 7B = 56GB !
```

---

## 2.5 为什么Adam状态必须FP32？

```
m和v是累积量，需要高精度:

1. 动量m累积梯度历史
   - 每步更新: m = 0.9 * m + 0.1 * grad
   - 长期累积，小数值很重要

2. 方差v累积平方梯度
   - 决定每个参数的学习率缩放
   - 需要精确跟踪变化

3. 实验表明:
   - FP16 Adam状态 → 训练发散
   - 8-bit Adam → 可行但需要特殊处理
```

## 2.6 激活值内存详解

```python
def estimate_activations(batch, seq, hidden, layers, heads, ffn_mult=4):
    """
    详细激活值估算
    """
    head_dim = hidden // heads

    activations_per_layer = {
        # Attention部分
        'qkv_proj_input': batch * seq * hidden,           # 输入
        'q': batch * heads * seq * head_dim,              # Q
        'k': batch * heads * seq * head_dim,              # K
        'v': batch * heads * seq * head_dim,              # V
        'attention_scores': batch * heads * seq * seq,    # QK^T
        'attention_probs': batch * heads * seq * seq,     # softmax后
        'attention_output': batch * seq * hidden,         # PV

        # FFN部分
        'ffn_input': batch * seq * hidden,
        'ffn_hidden': batch * seq * hidden * ffn_mult,    # 最大的!
        'ffn_output': batch * seq * hidden,
    }

    total_per_layer = sum(activations_per_layer.values())
    total = total_per_layer * layers * 2  # FP16 = 2 bytes

    return total / 1e9  # GB

# 7B: batch=32, seq=2048, hidden=4096, layers=32, heads=32
print(f"激活值: {estimate_activations(32, 2048, 4096, 32, 32):.1f} GB")
# 约 48 GB
```

## 3.1 推理内存组成

```
推理不需要:
❌ 梯度
❌ 优化器状态
❌ 训练激活值

推理需要:
✅ 模型权重
✅ KV Cache (关键!)
✅ 少量运行时buffer

推理内存 = 权重 + KV Cache + Buffer
```

---

## 3.2 为什么需要KV Cache？

```python
# 自回归生成的问题
def generate_without_cache(model, prompt, max_new_tokens):
    tokens = prompt
    for _ in range(max_new_tokens):
        # 每次都要处理所有token!
        logits = model(tokens)  # O(n²) attention
        next_token = sample(logits[:, -1, :])
        tokens = torch.cat([tokens, next_token], dim=-1)
    return tokens

# 第1个token: 处理 n 个token
# 第2个token: 处理 n+1 个token
# ...
# 第k个token: 处理 n+k-1 个token
# 总计: n + (n+1) + ... + (n+k-1) = O(nk + k²)

# 使用KV Cache: 每步只需处理1个新token
# 总计: O(n + k)
```

---

## 3.3 KV Cache工作原理

```python
def generate_with_cache(model, prompt, max_new_tokens):
    # 初始化: 处理完整prompt
    kv_cache = {}
    tokens = prompt
    logits, kv_cache = model(tokens, kv_cache)  # 填充cache

    generated = []
    for _ in range(max_new_tokens):
        # 只处理最后一个token!
        last_token = generated[-1] if generated else tokens[:, -1:]
        logits, kv_cache = model(last_token, kv_cache)
        # kv_cache自动追加新的K, V

        next_token = sample(logits)
        generated.append(next_token)

    return torch.cat([tokens] + generated, dim=-1)

# KV Cache存储:
# - 每层的K: [batch, num_kv_heads, seq_len, head_dim]
# - 每层的V: [batch, num_kv_heads, seq_len, head_dim]
```

---

## 3.4 KV Cache大小公式

```python
def kv_cache_size(batch, seq_len, layers, num_kv_heads, head_dim, dtype_bytes=2):
    """
    KV Cache = 2 × layers × batch × kv_heads × seq × head_dim × bytes

    因子解释:
    - 2: K和V各一份
    - layers: 每层都有独立的KV
    - batch: 每个样本独立
    - kv_heads: GQA中的KV head数
    - seq: 序列长度
    - head_dim: 每个head的维度
    - bytes: 数据类型大小
    """
    return 2 * layers * batch * num_kv_heads * seq_len * head_dim * dtype_bytes / 1e9

# LLaMA-7B: 32层, 32 heads, head_dim=128
print(f"seq=2K: {kv_cache_size(1, 2048, 32, 32, 128):.2f} GB")   # 1.0 GB
print(f"seq=8K: {kv_cache_size(1, 8192, 32, 32, 128):.2f} GB")   # 4.0 GB
print(f"seq=32K: {kv_cache_size(1, 32768, 32, 32, 128):.2f} GB") # 16.0 GB
print(f"seq=128K: {kv_cache_size(1, 131072, 32, 32, 128):.2f} GB") # 64.0 GB
```

## 3.6 GQA: 减少KV Cache

```python
# MHA vs GQA

# MHA (Multi-Head Attention): num_kv_heads = num_heads
# 每个query head有独立的k, v head
# LLaMA-7B MHA: 32个KV heads

# GQA (Grouped Query Attention): num_kv_heads < num_heads
# 多个query heads共享k, v heads
# LLaMA-2 70B GQA: 8个KV heads, 64个Q heads

def compare_kv_cache(seq_len=4096, layers=80, head_dim=128):
    # 70B MHA (假设)
    mha = kv_cache_size(1, seq_len, layers, 64, head_dim)

    # 70B GQA (实际)
    gqa = kv_cache_size(1, seq_len, layers, 8, head_dim)

    print(f"MHA: {mha:.1f} GB")  # 20.5 GB
    print(f"GQA: {gqa:.1f} GB")   # 2.6 GB
    print(f"节省: {(1 - gqa/mha)*100:.0f}%")  # 87%

compare_kv_cache()
```

## 4.3 Adafactor

```python
# Adafactor: 几乎零额外内存!

from transformers import Adafactor

optimizer = Adafactor(
    model.parameters(),
    lr=1e-3,
    relative_step=False,  # 使用固定学习率
    warmup_init=False
)

# 原理:
# 不存储m和v，而是:
# 1. 对行和列分别维护统计量
# 2. m, v 从 O(n×m) 降到 O(n+m)

# 例如: 4096×4096 矩阵
# Adam: 4096×4096×2×4 = 128MB
# Adafactor: (4096+4096)×2×4 = 64KB

# 缺点:
# - 收敛可能略慢
# - 需要调整学习率
```

## 4.4 Gradient Checkpointing

```python
# 原理: 不存储中间激活，反向传播时重新计算

# 启用方式
model.gradient_checkpointing_enable()

# 或手动控制
from torch.utils.checkpoint import checkpoint

class TransformerBlock(nn.Module):
    def forward(self, x):
        # 这个块的激活不会被存储
        x = checkpoint(self.attention, x)
        x = checkpoint(self.ffn, x)
        return x

# 效果 (7B, batch=32):
# 无checkpoint: 激活 48GB
# 有checkpoint: 激活 ~10GB

# 代价:
# - 计算增加 ~30% (需要重算前向)
# - 训练时间增加 ~20%
```

## 6.1 核心公式总结

```
训练内存 (混合精度 + Adam):
Memory = 16P + Activations
其中: 16 = 4(FP32主) + 2(FP16) + 2(梯度) + 8(Adam)

推理内存:
Memory = 2P + KV_Cache

KV Cache:
KV = 2 × L × B × H_kv × S × D × bytes
```

## Q&A

### Q1: 为什么Adam必须FP32？
**A**: m和v是累积量，FP16精度不够会累积舍入误差。8-bit可行是因为使用了特殊的block-wise量化。

### Q2: ZeRO-2和ZeRO-3怎么选？
**A**: ZeRO-2通信开销小，内存够用优先选。ZeRO-3内存节省更多但通信增加约50%。

### Q3: GQA会影响模型质量吗？
**A**: 研究表明8个KV heads和64个KV heads效果相近，但内存减少8倍。

### Q4: 训练时激活值能量化吗？
**A**: 可以用FP8激活，但需要特殊硬件(H100)。普通情况下激活保持FP16。

## 附录: 内存计算器

```python
def memory_calculator(
    params_b: float,        # 参数量 (billions)
    batch_size: int,
    seq_len: int,
    hidden_dim: int,
    num_layers: int,
    num_heads: int,
    num_kv_heads: int = None,  # GQA
    dtype_bytes: int = 2,   # FP16
    optimizer: str = "adam",  # adam, adam8bit, adafactor
    checkpointing: bool = False,
    mode: str = "train"     # train, inference
):
    """完整内存计算器"""
    P = params_b * 1e9

    if mode == "inference":
        # 推理: 权重 + KV Cache
        weights = P * dtype_bytes / 1e9

        kv_heads = num_kv_heads or num_heads
        head_dim = hidden_dim // num_heads
        kv_cache = kv_cache_size(batch_size, seq_len, num_layers,
                                 kv_heads, head_dim, dtype_bytes)

        return {
            "weights": weights,
            "kv_cache": kv_cache,
            "total": weights + kv_cache
        }

    else:
        # 训练
        fp32_master = P * 4 / 1e9
        fp16_working = P * 2 / 1e9
        gradients = P * 2 / 1e9

        if optimizer == "adam":
            opt_states = P * 8 / 1e9
        elif optimizer == "adam8bit":
            opt_states = P * 2 / 1e9
        else:  # adafactor
            opt_states = 0

        # 简化激活估算
        activations = estimate_activations(
            batch_size, seq_len, hidden_dim,
            num_layers, num_heads
        )
        if checkpointing:
            activations *= 0.2

        return {
            "fp32_master": fp32_master,
            "fp16_working": fp16_working,
            "gradients": gradients,
            "optimizer": opt_states,
            "activations": activations,
            "total": fp32_master + fp16_working + gradients + opt_states + activations
        }

# 使用示例
result = memory_calculator(
    params_b=7,
    batch_size=32,
    seq_len=2048,
    hidden_dim=4096,
    num_layers=32,
    num_heads=32,
    optimizer="adam",
    checkpointing=False,
    mode="train"
)
print(f"Total training memory: {result['total']:.1f} GB")
```
