def analyze_memory_usage():
    """分析不同配置的内存使用情况"""

    # 固定内存使用量估算
    def estimate_pinned_memory(batch_size, seq_len, hidden_dim):
        # 单个样本大小 (float32 = 4 bytes)
        sample_size = seq_len * hidden_dim * 4
        # 批次大小
        batch_memory = batch_size * sample_size
        # 预取缓冲区 (prefetch_factor * num_workers)
        buffer_memory = batch_memory * 2 * 4  # 假设prefetch=2, workers=4

        return {
            "sample_size_mb": sample_size / 1024**2,
            "batch_memory_mb": batch_memory / 1024**2,
            "buffer_memory_mb": buffer_memory / 1024**2,
            "total_pinned_mb": buffer_memory / 1024**2
        }

    return estimate_pinned_memory(32, 512, 768)

class OptimizedDataLoader:
    """优化的数据加载器包装类"""

    def __init__(self, dataset, batch_size, **kwargs):
        self.dataset = dataset
        self.batch_size = batch_size
        self.config = self._get_optimal_config(**kwargs)

    def _get_optimal_config(self, **kwargs):
        """根据系统资源自动优化配置"""

        # 系统资源检测
        cpu_count = os.cpu_count()
        memory_gb = psutil.virtual_memory().total / (1024**3)
        gpu_memory_gb = torch.cuda.get_device_properties(0).total_memory / (1024**3) if torch.cuda.is_available() else 0

        # 动态配置
        config = {
            "batch_size": self.batch_size,
            "num_workers": min(cpu_count, 8),
            "pin_memory": torch.cuda.is_available(),
            "persistent_workers": torch.cuda.is_available(),
            "prefetch_factor": 2 if memory_gb > 16 else 1
        }

        # 大内存系统可以更激进
        if memory_gb > 32:
            config["prefetch_factor"] = 3
            config["num_workers"] = min(cpu_count * 2, 12)

        # 更新用户自定义配置
        config.update(kwargs)
        return config

    def get_dataloader(self):
        return DataLoader(self.dataset, **self.config)

    def benchmark(self, num_batches=100):
        """性能基准测试"""
        dataloader = self.get_dataloader()

        start_time = time.time()
        for i, batch in enumerate(dataloader):
            if i >= num_batches:
                break
        end_time = time.time()

        throughput = num_batches / (end_time - start_time)
        print(f"数据加载吞吐量: {throughput:.1f} batches/sec")
        return throughput

## 总结

### pin_memory 核心要点
- ✅ GPU训练时必须开启，显著提升数据传输速度
- ❌ CPU训练时不要开启，浪费内存资源
- ⚠️ 增加系统内存压力，需要监控系统内存使用

### persistent_workers 核心要点
- ✅ 多进程训练时开启，消除epoch间延迟
- ❌ num_workers=0时无效
- ⚠️ 需要正确处理资源清理，避免内存泄漏
- 🎯 配合prefetch_factor使用效果最佳

## 📐 数学形式化分析

### 1. 数据加载吞吐量模型

#### 数据加载管道的数学建模

**定义1**: 数据加载管道的总时间包含：

$$T_{total} = T_{read} + T_{transform} + T_{transfer}$$

其中：
- $T_{read}$: 从存储读取数据时间
- $T_{transform}$: 数据预处理时间（如解码、增强）
- $T_{transfer}$: CPU到GPU传输时间

#### 多进程加速的理论分析

**定理1**: 使用 $n$ 个worker时，理论加速比为：

$$\text{Speedup}_{workers} = \min\left(n, \frac{T_{total}}{T_{bottleneck}}\right)$$

其中 $T_{bottleneck}$ 是瓶颈操作时间（通常是磁盘I/O）。

**证明**:
- 理想情况：$n$ 个worker并行处理，总时间降为 $\frac{T_{total}}{n}$
- 实际情况：受限于瓶颈资源（如磁盘带宽），无法完全线性加速
- 因此实际加速比受瓶颈限制

### 2. pin_memory 的数学分析

#### 内存传输时间模型

**定理2**: pin_memory对传输时间的影响：

$$T_{transfer} = \begin{cases}
T_{pageable} = \frac{D}{B_{system}} + T_{overhead} & \text{without pin\_memory} \\
T_{pinned} = \frac{D}{B_{DMA}} & \text{with pin\_memory}
\end{cases}$$

其中：
- $D$: 数据大小（字节）
- $B_{system}$: 系统内存带宽
- $B_{DMA}$: DMA直接内存访问带宽
- $T_{overhead}$: 页面锁定开销

**加速比**:
$$\text{Speedup}_{pin} = \frac{T_{pageable}}{T_{pinned}} \approx 2-5\times$$

### 3. prefetch_factor 的优化分析

#### 预取缓冲的数学模型

**定义2**: 预取因子 $p$ 控制预取的batch数量：

$$N_{prefetch} = p \times n_{workers}$$

**最优预取因子**:

$$p^* = \arg\min_p \left(T_{wait}(p) + M_{memory}(p)\right)$$

其中：
- $T_{wait}(p)$: GPU等待数据的时间（$p$ 越大越小）
- $M_{memory}(p)$: 内存占用（$p$ 越大越大）

**权衡关系**: 
- $p$ 太小：GPU等待时间长
- $p$ 太大：内存占用过高
- 通常最优值：$p \in [2, 4]$

### 4. persistent_workers 的性能模型

#### Worker重启开销分析

**定理3**: persistent_workers减少的开销为：

$$\Delta T = (N_{epochs} - 1) \times n_{workers} \times T_{startup}$$

其中：
- $N_{epochs}$: epoch数量
- $n_{workers}$: worker数量
- $T_{startup}$: 单个worker启动时间（通常 0.5-2秒）

**节省比例**:
$$\text{Saving} = \frac{\Delta T}{T_{total\_training}}$$

对于长时间训练，节省可达 5-10%。

验证多进程加速、pin_memory效果、prefetch优化等

import time
import torch
import numpy as np
from torch.utils.data import DataLoader, Dataset
import matplotlib.pyplot as plt
from typing import Dict, List

class DummyDataset(Dataset):
    """模拟数据集"""
    def __init__(self, size: int = 10000, data_shape: tuple = (3, 224, 224)):
        self.size = size
        self.data_shape = data_shape
    
    def __len__(self):
        return self.size
    
    def __getitem__(self, idx):
        # 模拟数据读取和预处理
        time.sleep(0.001)  # 模拟I/O延迟
        data = torch.randn(*self.data_shape)
        label = torch.randint(0, 1000, (1,)).item()
        return data, label

class DataLoaderPerformanceAnalyzer:
    """数据加载器性能分析器"""
    
    def benchmark_num_workers(
        self,
        dataset: Dataset,
        batch_size: int = 32,
        num_workers_list: List[int] = [0, 1, 2, 4, 8]
    ) -> Dict[str, List]:
        """
        测试不同worker数量的性能
        
        Returns:
            性能基准测试结果
        """
        results = {
            'num_workers': [],
            'time_per_epoch': [],
            'throughput': [],
            'theoretical_speedup': [],
            'actual_speedup': []
        }
        
        baseline_time = None
        
        for num_workers in num_workers_list:
            dataloader = DataLoader(
                dataset,
                batch_size=batch_size,
                num_workers=num_workers,
                pin_memory=False,
                persistent_workers=False if num_workers == 0 else True
            )
            
            # 预热
            for i, _ in enumerate(dataloader):
                if i >= 5:
                    break
            
            # 测试
            start = time.perf_counter()
            batch_count = 0
            for batch in dataloader:
                batch_count += 1
                if batch_count >= 100:  # 测试100个batch
                    break
            elapsed = time.perf_counter() - start
            
            if baseline_time is None:
                baseline_time = elapsed
            
            results['num_workers'].append(num_workers)
            results['time_per_epoch'].append(elapsed)
            results['throughput'].append(batch_count / elapsed)
            results['theoretical_speedup'].append(num_workers if num_workers > 0 else 1)
            results['actual_speedup'].append(baseline_time / elapsed)
        
        return results
    
    def benchmark_pin_memory(
        self,
        dataset: Dataset,
        batch_size: int = 32,
        num_workers: int = 4
    ) -> Dict[str, float]:
        """
        测试pin_memory的效果
        
        Returns:
            pin_memory性能对比
        """
        results = {}
        
        for pin_memory in [False, True]:
            dataloader = DataLoader(
                dataset,
                batch_size=batch_size,
                num_workers=num_workers,
                pin_memory=pin_memory,
                persistent_workers=True
            )
            
            # 预热
            for i, batch in enumerate(dataloader):
                if i >= 5:
                    break
                if torch.cuda.is_available():
                    _ = batch[0].cuda()
            
            # 测试
            start = time.perf_counter()
            batch_count = 0
            for batch in dataloader:
                if torch.cuda.is_available():
                    _ = batch[0].cuda()
                batch_count += 1
                if batch_count >= 100:
                    break
            elapsed = time.perf_counter() - start
            
            key = f'pin_memory={pin_memory}'
            results[key] = elapsed
        
        # 计算加速比
        speedup = results['pin_memory=False'] / results['pin_memory=True']
        results['speedup'] = speedup
        
        return results
    
    def benchmark_prefetch_factor(
        self,
        dataset: Dataset,
        batch_size: int = 32,
        num_workers: int = 4,
        prefetch_factors: List[int] = [1, 2, 4, 8]
    ) -> Dict[str, List]:
        """
        测试不同prefetch_factor的效果
        
        Returns:
            prefetch_factor性能分析
        """
        results = {
            'prefetch_factor': [],
            'time': [],
            'memory_mb': []
        }
        
        for prefetch in prefetch_factors:
            dataloader = DataLoader(
                dataset,
                batch_size=batch_size,
                num_workers=num_workers,
                pin_memory=True,
                persistent_workers=True,
                prefetch_factor=prefetch
            )
            
            # 预热
            for i, _ in enumerate(dataloader):
                if i >= 5:
                    break
            
            # 测试
            import psutil
            process = psutil.Process()
            mem_before = process.memory_info().rss / 1024 / 1024
            
            start = time.perf_counter()
            batch_count = 0
            for _ in dataloader:
                batch_count += 1
                if batch_count >= 100:
                    break
            elapsed = time.perf_counter() - start
            
            mem_after = process.memory_info().rss / 1024 / 1024
            
            results['prefetch_factor'].append(prefetch)
            results['time'].append(elapsed)
            results['memory_mb'].append(mem_after - mem_before)
        
        return results
    
    def verify_persistent_workers_savings(
        self,
        dataset: Dataset,
        batch_size: int = 32,
        num_workers: int = 4,
        num_epochs: int = 5
    ) -> Dict[str, float]:
        """
        验证persistent_workers的节省时间
        
        Returns:
            节省时间分析
        """
        results = {}
        
        for persistent in [False, True]:
            total_time = 0
            
            for epoch in range(num_epochs):
                dataloader = DataLoader(
                    dataset,
                    batch_size=batch_size,
                    num_workers=num_workers,
                    persistent_workers=persistent
                )
                
                start = time.perf_counter()
                batch_count = 0
                for _ in dataloader:
                    batch_count += 1
                    if batch_count >= 20:  # 每个epoch 20个batch
                        break
                elapsed = time.perf_counter() - start
                total_time += elapsed
            
            key = f'persistent={persistent}'
            results[key] = total_time
        
        # 计算节省
        savings = results['persistent=False'] - results['persistent=True']
        savings_percent = savings / results['persistent=False'] * 100
        
        results['savings_seconds'] = savings
        results['savings_percent'] = savings_percent
        
        return results
