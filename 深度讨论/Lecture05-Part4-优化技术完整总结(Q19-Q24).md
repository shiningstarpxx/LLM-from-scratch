### 限制因素分析

#### 多维度约束

```python
def calculate_occupancy(
    threads_per_block: int,
    registers_per_thread: int,
    shared_mem_per_block: int
):
    """
    计算Occupancy的完整函数
    """
    # 1. Warp限制
    warps_per_block = threads_per_block // 32
    max_blocks_warp = 64 // warps_per_block
    
    # 2. 寄存器限制
    regs_per_block = threads_per_block * registers_per_thread
    max_blocks_reg = 65536 // regs_per_block
    
    # 3. Shared Memory限制
    max_blocks_smem = (164*1024) // shared_mem_per_block
    
    # 4. 物理限制 (A100: max 16 blocks/SM)
    max_blocks_physical = 16
    
    # 实际能跑的blocks (取最小值)
    actual_blocks = min(
        max_blocks_warp,
        max_blocks_reg,
        max_blocks_smem,
        max_blocks_physical
    )
    
    # Occupancy
    actual_warps = actual_blocks * warps_per_block
    occupancy = actual_warps / 64
    
    return occupancy, {
        'bottleneck': identify_bottleneck(),
        'actual_blocks': actual_blocks,
        'actual_warps': actual_warps
    }
```

### 核心机制

#### 1. Divergence执行过程

```c
__global__ void divergent_kernel(int* data) {
    int x = data[threadIdx.x];
    
    if (x > 0) {          // 分支！
        // Path A: 一半线程走这里
        x = x * 2;
    } else {
        // Path B: 另一半线程走这里
        x = x + 1;
    }
}

执行过程:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. 计算条件 (x > 0) for 所有32线程
2. Path A: 执行 x*2, Path B线程闲置 ⚠️
3. Path B: 执行 x+1, Path A线程闲置 ⚠️
4. 汇合

实际时间 = Time(Path A) + Time(Path B)
vs 理想 = max(Time(Path A), Time(Path B))

性能损失 ~ 50% (如果分支均匀)
学员说的"2倍计算量"完全正确！⭐⭐⭐⭐⭐
```

#### 2. 量化分析

```python
def divergence_cost(
    branch_probability: float,
    instructions_true: int,
    instructions_false: int
):
    """
    计算Divergence的性能损失
    """
    # 无分支（理想情况）
    no_div_time = max(instructions_true, instructions_false)
    
    # 有分支（串行执行）
    with_div_time = instructions_true + instructions_false
    
    # 性能损失
    slowdown = with_div_time / no_div_time
    
    return slowdown

# 例子
slowdown = divergence_cost(
    branch_probability=0.5,
    instructions_true=10,
    instructions_false=10
)
# → slowdown = 2x ⚠️⚠️
# 学员的"2倍计算量"完全正确！
```

```c
// ❌ 有Divergence
__global__ void divergent(int* data) {
    int x = data[threadIdx.x];

        x = x * 2;

        x = x + 1;
    }

// ✅ 无Divergence (Predication)
__global__ void no_divergence(int* data) {
    int x = data[threadIdx.x];
    int mask = x > 0;
    
    // 全部线程都执行，但结果有条件地写入
    int result_a = x * 2;
    int result_b = x + 1;
    int result = mask ? result_a : result_b;
    
    data[threadIdx.x] = result;
}
```

```python
# 将满足条件的data放一起
```

### 硬件-软件协同

#### 1. 完整的混合精度流程

```python
# 软件层面 (PyTorch)

scaler = GradScaler()

with autocast():  # FP16前向
    output = model(input)
    loss = criterion(output, target)

scaler.scale(loss).backward()  # Loss scaling
scaler.step(optimizer)
scaler.update()

# 硬件层面
# Tensor Cores自动执行FP16×FP16 + FP32累加
# → 快速且精确
# GPU硬件自动路由到Tensor/CUDA Cores
# 不是CPU决定，是GPU硬件自动选择！
```

#### 2. Loss Scaling机制

```python
Loss Scaling的硬件原因:
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

FP16表示范围: 2^-24 ~ 2^15
梯度通常很小: ~10^-7

问题:
小梯度 → FP16 underflow → 变成0
→ 训练失败 ❌

解决方案: Loss Scaling
loss *= 1024  # 放大loss
backward()    # 梯度也放大1024倍
grads /= 1024 # 缩小回来

效果:
梯度从10^-7 → 10^-4
→ FP16可表示 ✅

硬件视角:
这是利用FP16的动态范围
不需要硬件改动
学员的"防止溢出"方向正确，更准确是"防止下溢"
```

#### 3. 自动混合精度 (AMP)

```python

    'matmul',      # Tensor Cores!

    'softmax',     # 需要高精度
    'layer_norm',  # 需要高精度
```
