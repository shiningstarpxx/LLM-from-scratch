## 🎯 量化 vs 精度转换：本质区别

### 基本概念对比

```python
# === 精度转换 ===
# FP32 → FP16：保持浮点特性，降低精度
fp32_value = 3.14159265359
fp16_value = torch.float16(fp32_value)  # 3.140625
print(f"精度转换: {fp32_value} → {fp16_value}")
print("特性: 仍是浮点数，动态范围缩小，精度降低")

# === 量化 ===
# FP32 → INT8：映射到整数，完全不同的数值系统
def quantize_to_int8(value, scale, zero_point):
    """量化函数：FP32 → INT8"""
    return int(round(value / scale + zero_point))

# 示例：量化范围 [-1.0, 1.0] 到 INT8 [-128, 127]
scale = 2.0 / 255  # 缩放因子
zero_point = 128   # 零点

fp32_value = 3.14159265359
clipped_value = max(-1.0, min(1.0, fp32_value))  # 限制到量化范围
int8_value = quantize_to_int8(clipped_value, scale, zero_point)

print(f"量化: {fp32_value} → {int8_value}")
print("特性: 映射到整数，有限离散值，需要反量化恢复")
```

### 核心区别总结

| 特性 | 精度转换 (FP32→FP16) | 量化 (FP32→INT8) |
|------|---------------------|------------------|
| **数值类型** | 浮点数 → 浮点数 | 浮点数 → 整数 |
| **动态范围** | 缩小 (10^38 → 10^4) | 有限离散值 |
| **精度损失** | 精度降低 | 范围压缩 + 离散化 |
| **硬件支持** | 原生支持 | 需要专用指令 |
| **内存节省** | 50% | 75% |
| **计算加速** | 有限 | 显著 |

## 🔬 量化的数学原理

### 1. 线性量化公式

```python
def linear_quantize(tensor, dtype=torch.qint8):
    """线性量化的完整实现"""

    # 确定量化范围
    if dtype == torch.qint8:
        qmin, qmax = -128, 127
    elif dtype == torch.quint8:
        qmin, qmax = 0, 255

    # 计算缩放因子和零点
    min_val, max_val = tensor.min().item(), tensor.max().item()

    # 防止除零
    if max_val == min_val:
        scale = 1.0
        zero_point = 0
    else:
        scale = (max_val - min_val) / (qmax - qmin)
        zero_point = qmin - round(min_val / scale)

        # 确保零点在有效范围内
        zero_point = max(qmin, min(qmax, zero_point))

    # 量化操作
    quantized = torch.round(tensor / scale + zero_point)
    quantized = torch.clamp(quantized, qmin, qmax)

    return quantized.to(torch.int8), scale, zero_point

def linear_dequantize(quantized, scale, zero_point):
    """反量化操作"""
    return (quantized.float() - zero_point) * scale

# 示例演示
if __name__ == "__main__":
    # 创建测试张量
    original = torch.randn(1000) * 2 + 1  # 均值1，标准差2

    # 量化
    quantized, scale, zero_point = linear_quantize(original)

    # 反量化
    dequantized = linear_dequantize(quantized, scale, zero_point)

    # 计算误差
    mse_error = torch.mean((original - dequantized) ** 2).item()

    print(f"原始范围: [{original.min():.3f}, {original.max():.3f}]")
    print(f"量化参数: scale={scale:.6f}, zero_point={zero_point}")
    print(f"量化后范围: [{quantized.min()}, {quantized.max()}]")
    print(f"反量化范围: [{dequantized.min():.3f}, {dequantized.max():.3f}]")
    print(f"均方误差: {mse_error:.6f}")
```

### 2. 对称量化 vs 非对称量化

```python
def symmetric_quantize(tensor, bits=8):
    """对称量化：零点固定为0"""

    # 计算最大绝对值
    max_abs = torch.max(torch.abs(tensor)).item()

    # 确定量化范围
    if bits == 8:
        qmax = 127  # INT8范围 [-127, 127]
    elif bits == 4:
        qmax = 7    # INT4范围 [-7, 7]

    # 计算缩放因子（零点为0）
    scale = max_abs / qmax if max_abs > 0 else 1.0

    # 量化
    quantized = torch.round(tensor / scale)
    quantized = torch.clamp(quantized, -qmax, qmax)

    return quantized.to(torch.int8), scale

def asymmetric_quantize(tensor, bits=8):
    """非对称量化：零点可调整"""

    # 确定量化范围
    if bits == 8:
        qmin, qmax = -128, 127
    elif bits == 4:
        qmin, qmax = -8, 7

    min_val, max_val = tensor.min().item(), tensor.max().item()

    # 计算缩放因子和零点
    scale = (max_val - min_val) / (qmax - qmin) if max_val != min_val else 1.0
    zero_point = qmin - round(min_val / scale)
    zero_point = max(qmin, min(qmax, zero_point))

    # 量化
    quantized = torch.round(tensor / scale + zero_point)
    quantized = torch.clamp(quantized, qmin, qmax)

    return quantized.to(torch.int8), scale, zero_point

# 对比演示
def compare_quantization_methods():
    """对比对称和非对称量化"""

    # 创建非对称分布的张量
    tensor = torch.randn(1000) + 2  # 均值2，偏向正值

    # 对称量化
    sym_quant, sym_scale = symmetric_quantize(tensor)
    sym_dequant = sym_quant.float() * sym_scale

    # 非对称量化
    asym_quant, asym_scale, asym_zp = asymmetric_quantize(tensor)
    asym_dequant = (asym_quant.float() - asym_zp) * asym_scale

    # 计算误差
    sym_error = torch.mean((tensor - sym_dequant) ** 2).item()
    asym_error = torch.mean((tensor - asym_dequant) ** 2).item()

    print(f"原始张量范围: [{tensor.min():.3f}, {tensor.max():.3f}]")
    print(f"对称量化误差: {sym_error:.6f}")
    print(f"非对称量化误差: {asym_error:.6f}")
    print(f"非对称量化优势: {'是' if asym_error < sym_error else '否'}")

compare_quantization_methods()
```

```python
def accuracy_loss_analysis():
    """量化精度损失分析"""

    # 创建测试模型
    model = SimpleModel()
    model.eval()

    # 测试数据
    test_data = torch.randn(1000, 784)

    # 原始模型预测
    with torch.no_grad():
        original_output = model(test_data)

    # 量化模型预测
    quantized_model = quant.quantize_dynamic(model, {nn.Linear}, dtype=torch.qint8)
    with torch.no_grad():
        quantized_output = quantized_model(test_data)

    # 计算预测差异
    mse_loss = nn.MSELoss()(original_output, quantized_output.float()).item()

    # 计算预测一致性
    original_pred = torch.argmax(original_output, dim=1)
    quantized_pred = torch.argmax(quantized_output, dim=1)
    accuracy_consistency = (original_pred == quantized_pred).float().mean().item()

    print("=== 精度损失分析 ===")
    print(f"MSE损失: {mse_loss:.6f}")
    print(f"预测一致性: {accuracy_consistency*100:.2f}%")
    print(f"预测不一致性: {(1-accuracy_consistency)*100:.2f}%")

    # 分析不同层的量化影响
    print("\n=== 逐层分析 ===")
    for name, module in model.named_modules():
        if isinstance(module, nn.Linear):
            weight = module.weight.data
            quantized_weight = quant.quantize_dynamic(
                module, {nn.Linear}, dtype=torch.qint8
            ).fc1.weight().data

            weight_error = torch.mean((weight.float() - quantized_weight.float()) ** 2).item()
            print(f"{name}: 权重量化误差 = {weight_error:.6f}")

    return mse_loss, accuracy_consistency
```

## 📐 数学形式化证明

### 1. 量化作为线性变换的数学证明

#### 定理1: 仿射量化函数

**量化函数**:
$$Q(r) = \text{clip}\left(\left\lfloor \frac{r}{s} \right\rceil + z, q_{min}, q_{max}\right)$$

其中：
- $r \in \mathbb{R}$: 实数值
- $s \in \mathbb{R}^+$: 缩放因子(scale)
- $z \in \mathbb{Z}$: 零点(zero-point)
- $\lfloor \cdot \rceil$: 四舍五入
- $\text{clip}$: 截断到 $[q_{min}, q_{max}]$

**反量化函数**:
$$\tilde{r} = s \times (q - z)$$

**性质**: 量化是仿射变换 + 离散化 + 截断。

#### 定理2: 最优缩放因子和零点

**目标**: 最小化量化误差（MSE）

$$\min_{s, z} \mathbb{E}[(r - \tilde{r})^2] = \min_{s, z} \mathbb{E}[(r - s(Q(r) - z))^2]$$

**解析解**（对称量化，$z=0$）:
$$s^* = \frac{\max(|r_{min}|, |r_{max}|)}{2^{b-1} - 1}$$

**解析解**（非对称量化）:
$$s^* = \frac{r_{max} - r_{min}}{q_{max} - q_{min}}$$
$$z^* = \text{round}\left(q_{min} - \frac{r_{min}}{s^*}\right)$$

### 2. 量化误差分析

#### 量化误差的分布

**定理3**: 在均匀量化下，量化误差服从均匀分布。

**假设**: $r$ 在 $[a, b]$ 上均匀分布，量化间隔为 $\Delta = s$。

**量化误差**: 
$$e = r - \tilde{r} = r - s\lfloor r/s \rceil$$

**误差范围**: $e \in [-\Delta/2, \Delta/2]$

**期望误差**: 
$$\mathbb{E}[e] = 0$$

**方差**:
$$\text{Var}[e] = \frac{\Delta^2}{12} = \frac{s^2}{12}$$

#### 信噪比(SNR)

**定义**: 量化后的信噪比

$$\text{SNR} = 10 \log_{10}\left(\frac{\sigma_r^2}{\sigma_e^2}\right)$$

其中 $\sigma_r^2$ 是信号方差，$\sigma_e^2$ 是量化误差方差。

**对于b位量化**:
$$\sigma_e^2 = \frac{1}{12}\left(\frac{r_{max} - r_{min}}{2^b - 1}\right)^2$$

**SNR近似**:
$$\text{SNR} \approx 6.02b + 1.76 \text{ (dB)}$$

即：每增加1位，SNR增加约6dB！

### 3. 量化感知训练(QAT)的数学原理

#### Straight-Through Estimator (STE)

**问题**: 量化函数 $Q$ 不可微。

**前向传播**: 
$$\tilde{W} = Q(W)$$

**反向传播**: 近似梯度
$$\frac{\partial \mathcal{L}}{\partial W} \approx \frac{\partial \mathcal{L}}{\partial \tilde{W}} \times 1$$

即：**假装量化函数是恒等映射**！

**数学不严格，但实践有效。**

#### Gumbel-Softmax量化

**更严格的连续松弛**:

$$\tilde{q}_i = \sum_{k=q_{min}}^{q_{max}} p_k(r) \times k$$

其中 $p_k(r)$ 是选择量化级别 $k$ 的概率。

**Gumbel-Softmax**:
$$p_k(r) = \frac{\exp((h_k(r) + G_k)/\tau)}{\sum_j \exp((h_j(r) + G_j)/\tau)}$$

其中 $G_k \sim \text{Gumbel}(0, 1)$, $\tau$ 是温度。

**性质**: $\tau \to 0$ 时退化为硬量化，但处处可微！

### 4. 混合精度量化的理论

#### 敏感度分析

**定义**: 层 $l$ 的敏感度 $\mathcal{S}_l$：

$$\mathcal{S}_l = \mathbb{E}\left[\left|\frac{\partial \mathcal{L}}{\partial W_l}\right|\right]$$

**策略**: 敏感度高的层使用高精度，敏感度低的层使用低精度。

#### Hessian引导的混合精度

**定理4**: 最优比特分配

**目标**: 在总比特预算 $B$ 下，最小化损失：

$$\min_{b_1, \ldots, b_L} \mathcal{L}(b_1, \ldots, b_L)$$

subject to: $\sum_{l=1}^L w_l b_l \leq B$

其中 $w_l$ 是层 $l$ 的参数量权重。

**二阶近似**（Hessian）:
$$\mathcal{L}(b) \approx \mathcal{L}(b_0) + \sum_l \frac{1}{2}\lambda_l(b_0) \times \frac{1}{2^{2(b_l - b_0)}}$$

其中 $\lambda_l$ 是层 $l$ 的Hessian最大特征值（敏感度）。

**最优解** (Lagrange乘数法):
$$b_l^* = b_0 + \frac{1}{2}\log_2\left(\frac{\lambda_l}{\lambda}\right)$$

其中 $\lambda$ 是满足总预算约束的Lagrange乘数。

### 5. INT8矩阵乘法的数学原理

#### 量化矩阵乘法

**问题**: 计算 $Y = WX$，其中 $W, X$ 是FP32。

**量化后**:
$$W_q = Q(W) = \text{round}(W / s_W) + z_W$$
$$X_q = Q(X) = \text{round}(X / s_X) + z_X$$

**INT8矩阵乘法**:
$$Y_q = W_q X_q$$

**反量化**:
$$(Y_q - z_Y) \times s_Y \approx (W_q - z_W)(X_q - z_X) \times s_W s_X$$

**展开**:
$$= W_q X_q - W_q z_X - z_W X_q + z_W z_X$$

**关键**: 需要校正项！

#### 高效实现

**预计算**:
- $\sum_i W_{qi}$（每行和）
- $\sum_j X_{qj}$（每列和）

**运行时**:
$$Y_{corrected} = (Y_q - \sum_i W_{qi} \times z_X - z_W \times \sum_j X_{qj} + z_W z_X) \times s_W s_X$$

### 6. 动态量化的在线算法

#### Exponential Moving Average (EMA)

**目标**: 在线估计动态范围。

**算法**:
$$\mu_t = \alpha \mu_{t-1} + (1-\alpha) \times \mathbb{E}[x_t]$$
$$\sigma_t^2 = \alpha \sigma_{t-1}^2 + (1-\alpha) \times \text{Var}[x_t]$$

**量化范围**:
$$[r_{min}, r_{max}] = [\mu_t - k\sigma_t, \mu_t + k\sigma_t]$$

其中 $k$ 通常取3（99.7%覆盖）。

**缩放因子**:
$$s_t = \frac{r_{max} - r_{min}}{2^b - 1}$$

#### Kahan累加提高精度

**问题**: 多次INT8累加可能溢出或精度损失。

**Kahan算法**:
```
sum = 0
c = 0  # 补偿
for value in values:
    y = value - c
    t = sum + y
    c = (t - sum) - y
    sum = t
```

**效果**: 减少浮点误差累积！

```python
"""
量化技术完整数学验证代码
验证量化误差、QAT、混合精度等

import numpy as np
import torch
import torch.nn as nn
import torch.nn.functional as F
import matplotlib.pyplot as plt
from typing import Dict, List, Tuple

class UniformQuantizer:
    """均匀量化器"""
    
    def __init__(
        self,
        num_bits: int = 8,
        symmetric: bool = False
    ):
        self.num_bits = num_bits
        self.symmetric = symmetric
        
        if symmetric:
            self.qmin = -(2 ** (num_bits - 1))
            self.qmax = 2 ** (num_bits - 1) - 1
        else:
            self.qmin = 0
            self.qmax = 2 ** num_bits - 1
    
    def calibrate(self, x: torch.Tensor) -> Tuple[float, float]:
        """校准缩放因子和零点"""
        xmin = x.min().item()
        xmax = x.max().item()
        
        if self.symmetric:
            xmax = max(abs(xmin), abs(xmax))
            xmin = -xmax
            zero_point = 0
        else:
            zero_point = self.qmin - round(xmin / ((xmax - xmin) / (self.qmax - self.qmin)))
            zero_point = max(self.qmin, min(self.qmax, zero_point))
        
        scale = (xmax - xmin) / (self.qmax - self.qmin)
        
        return scale, zero_point
    
    def quantize(
        self,
        x: torch.Tensor,
        scale: float,
        zero_point: float
    ) -> torch.Tensor:
        """量化"""
        x_int = torch.round(x / scale + zero_point)
        x_int = torch.clamp(x_int, self.qmin, self.qmax)
        return x_int.to(torch.int8)
    
    def dequantize(
        self,
        x_int: torch.Tensor,
        scale: float,
        zero_point: float
    ) -> torch.Tensor:
        """反量化"""
        return scale * (x_int.float() - zero_point)
    
    def compute_snr(
        self,
        original: torch.Tensor,
        quantized: torch.Tensor
    ) -> float:
        """计算信噪比"""
        signal_power = torch.mean(original ** 2)
        noise_power = torch.mean((original - quantized) ** 2)
        
        if noise_power == 0:
            return float('inf')
        
        snr_linear = signal_power / noise_power
        snr_db = 10 * torch.log10(snr_linear)
        
        return snr_db.item()


class QATLayer(nn.Module):
    """量化感知训练层"""
    
    def __init__(self, in_features: int, out_features: int, num_bits: int = 8):
        super().__init__()
        self.weight = nn.Parameter(torch.randn(out_features, in_features))
        self.bias = nn.Parameter(torch.zeros(out_features))
        
        self.quantizer = UniformQuantizer(num_bits, symmetric=True)
        self.num_bits = num_bits
        
        # 可学习的缩放因子
        self.register_buffer('scale', torch.tensor(1.0))
        self.register_buffer('zero_point', torch.tensor(0.0))
    
    def forward(self, x: torch.Tensor) -> torch.Tensor:
        if self.training:
            # QAT: 前向量化，反向直通
            scale, zero_point = self.quantizer.calibrate(self.weight)
            self.scale.copy_(scale)
            self.zero_point.copy_(zero_point)
            
            # 量化+反量化（前向）
            w_q = self.quantizer.quantize(self.weight, scale, zero_point)
            w_dq = self.quantizer.dequantize(w_q, scale, zero_point)
            
            # 梯度直通（自动微分会自动处理）
            return F.linear(x, w_dq, self.bias)
        else:
            # 推理：直接使用量化权重
            w_q = self.quantizer.quantize(self.weight, self.scale, self.zero_point)
            w_dq = self.quantizer.dequantize(w_q, self.scale, self.zero_point)
            return F.linear(x, w_dq, self.bias)


class MixedPrecisionQuantizer:
    """混合精度量化器"""
    
    def __init__(self, model: nn.Module):
        self.model = model
        self.sensitivities = {}
    
    def compute_sensitivity(
        self,
        dataloader: torch.utils.data.DataLoader,
        criterion: nn.Module
    ) -> Dict[str, float]:
        """计算每层的敏感度"""
        self.model.eval()
        
        for name, module in self.model.named_modules():
            if isinstance(module, nn.Linear):
                gradients = []
                
                for data, target in dataloader:
                    self.model.zero_grad()
                    output = self.model(data)
                    loss = criterion(output, target)
                    loss.backward()
                    
                    if module.weight.grad is not None:
                        gradients.append(module.weight.grad.abs().mean().item())
                
                self.sensitivities[name] = np.mean(gradients)
        
        return self.sensitivities
    
    def allocate_bits(
        self,
        total_bits_budget: float,
        min_bits: int = 2,
        max_bits: int = 8
    ) -> Dict[str, int]:
        """分配比特数"""
        if not self.sensitivities:
            raise ValueError("请先计算敏感度")
        
        # 按敏感度排序
        sorted_layers = sorted(
            self.sensitivities.items(),
            key=lambda x: x[1],
            reverse=True
        )
        
        # 简化分配：高敏感度层用高精度
        bit_allocation = {}
        for i, (name, sens) in enumerate(sorted_layers):
            # 线性插值
            ratio = i / len(sorted_layers)
            bits = max_bits - int(ratio * (max_bits - min_bits))
            bit_allocation[name] = bits
        
        return bit_allocation
```
